% ------- section styling (UNIT headers = full-width colour banner) -------
\newcommand{\secbanner}[1]{\colorbox{accent}{\parbox{\dimexpr\textwidth-2\fboxsep}{%
  \vspace{3pt}{\color{white}\Large\bfseries #1}\vspace{3pt}}}}
\titleformat{\section}{\Large\bfseries}{}{0pt}{\secbanner}
\titlespacing*{\section}{0pt}{18pt}{10pt}

\titleformat{\subsection}
  {\large\bfseries\color{accent}}{}{0pt}{}
  [\vspace{-7pt}{\color{accent}\rule{\textwidth}{1pt}}]
\titlespacing*{\subsection}{0pt}{14pt}{8pt}

\titleformat{\subsubsection}
  {\normalsize\bfseries\color{inkgray}}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{10pt}{4pt}
